from neuronxcc import nki
import neuronxcc.nki.language as nl

@nki.jit
def nki_add(a_tensor, b_tensor):
    # Initialize result array in HBM 
    result = nl.ndarray(a_tensor.shape, dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Trip count to handle tensors larger than 128 partition size
    trip_count = (a_tensor.shape[0] + nl.tile_size.pmax - 1) // nl.tile_size.pmax
    
    # Process in batches 
    for p in nl.affine_range(trip_count):
        # Generate indices
        i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)[:, None]
        i_f = nl.arange(a_tensor.shape[1])[None, :]
        
        # Load input tiles with masking for boundary conditions
        a_tile = nl.load(a_tensor[i_p, i_f], mask=(i_p < a_tensor.shape[0]))
        b_tile = nl.load(b_tensor[i_p, i_f], mask=(i_p < b_tensor.shape[0]))
        
        # Add tiles
        out_tile = nl.add(a_tile, b_tile)
        
        # Store result with masking
        nl.store(result[i_p, i_f], out_tile, mask=(i_p < a_tensor.shape[0]))
        
    return result
