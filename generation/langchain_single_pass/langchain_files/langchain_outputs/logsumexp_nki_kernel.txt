***I will implement a logsumexp kernel that efficiently processes tensor input along the last dimension using tiling for large tensors and NKI's numerical functions. The implementation will find max values for stability, subtract them, compute exponentials, sum them, take logarithm, and add the max values back.***

import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_logsumexp(a_tensor, dim=-1):
    # Handle default case (dim=-1) and negative dimensions
    ndim = len(a_tensor.shape)
    if dim < 0:
        dim = ndim + dim
    
    # We only support reduction along the last dimension for now
    if dim != ndim - 1:
        raise ValueError("Currently only supports reduction along the last dimension")
    
    # Get dimensions
    if ndim == 1:
        # For 1D tensor, reshape to 2D for consistent processing
        sz_p = 1
        sz_f = a_tensor.shape[0]
        result_shape = ()  # scalar result
    else:
        # For N-D tensor, we'll process as a 2D tensor
        # by collapsing all but the last dimension
        sz_p = 1
        for i in range(ndim - 1):
            sz_p *= a_tensor.shape[i]
        sz_f = a_tensor.shape[-1]
        result_shape = a_tensor.shape[:-1]
    
    # Initialize result array
    result = nl.ndarray(result_shape, dtype=a_tensor.dtype, buffer=nl.shared_hbm)

    trip_count = math.ceil(sz_p/nl.tile_size.pmax)

    for p in nl.affine_range(trip_count):

        i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
