import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_cumprod(a_tensor, dim=0):
    # Initialize result array with same shape and dtype as input
    result = nl.ndarray(a_tensor.shape, dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Get the size along the specified dimension

        dim = len(a_tensor.shape) + dim
    dim_size = a_tensor.shape[dim]
    
    # Handle 1D tensor case
    if len(a_tensor.shape) == 1:
        # Calculate the number of tiles needed
        trip_count = math.ceil(dim_size / nl.tile_size.pmax)
        
        # Process the tensor in tiles
        for i in nl.affine_range(trip_count):
            # Calculate indices for this tile
            start_idx = i * nl.tile_size.pmax
            end_idx = min(start_idx + nl.tile_size.pmax, dim_size)
            tile_size = end_idx - start_idx
            
            # Process each position in the cumulative product
            for j in nl.affine_range(tile_size):
                idx = start_idx + j

                    value = nl.load(a_tensor[0])
                else:
                    # Load the previous result and multiply by current value
                    prev_val = nl.load(result[idx-1])
                    curr_val = nl.load(a_tensor[idx])
                    value = nl.multiply(prev_val, curr_val)
