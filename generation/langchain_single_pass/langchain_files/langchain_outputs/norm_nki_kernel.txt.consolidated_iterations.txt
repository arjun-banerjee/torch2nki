import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_norm(a_tensor):
    # Get the shape of the input tensor
    vector_length = a_tensor.shape[0]
    
    # Initialize the result scalar 
    # For vector norm, the result is a scalar
    result = nl.zeros((), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Calculate the number of tiles needed to process the vector
    # due to hardware limitations (max partition size)
    trip_count = math.ceil(vector_length / nl.tile_size.pmax)
    
    # Initialize a temporary sum of squares variable
    sum_squares = nl.zeros((), dtype=a_tensor.dtype, buffer=nl.psum)
    
    # Process the vector in tiles
    for p in nl.affine_range(trip_count):
        # Generate indices for the current tile
        i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
        
        # Load input data from external memory to on-chip memory
        # Only load up to the actual size of the vector
        x_tile = nl.load(a_tensor[i_p], mask=(i_p < vector_length))
        
        # Square each element in the tile
        squared_tile = nl.square(x_tile)
        
        # Sum the squared values in this tile and accumulate
        sum_squares += nl.sum(squared_tile, axis=0, mask=(i_p < vector_length))
    
    # Take the square root of the sum of squares to get the norm
    norm_result = nl.sqrt(sum_squares)
    
    # Store the result in the output variable
    nl.store(result, norm_result)
    
    return result

import neuronxcc.nki.language as nl

@nki.jit
def nki_norm(a_tensor):
    # Get the shape of the input tensor
    vector_length = a_tensor.shape[0]
    
    # Initialize a scalar to hold the sum of squares
    # We use psum buffer for accumulation
    sum_of_squares = nl.zeros((), dtype=nl.float32, buffer=nl.psum)
    
    # Calculate the number of tiles needed
    trip_count = math.ceil(vector_length / nl.tile_size.pmax)
    
    # Process the vector in tiles to respect hardware limitations
    for p in nl.affine_range(trip_count):

        i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
        
        # Load input data from external memory to on-chip memory
        # Only load up to the actual size of the tensor
        x_tile = nl.load(a_tensor[i_p], mask=(i_p < vector_length))
        
        # Square the elements
        squared_tile = nl.square(x_tile)
        
        # Sum the squares for this tile
        tile_sum = nl.sum(squared_tile, axis=0)
        
        # Accumulate into the running sum
        sum_of_squares += tile_sum
    
    # Take the square root of the sum of squares
    norm_result = nl.sqrt(sum_of_squares)
    
    # Initialize the result with the same dtype as input
    result = nl.ndarray((), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Store the final result
    nl.store(result, value=norm_result)
    
    return result

from neuronxcc import nki
import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_norm(a_tensor):
    # Get the shape of the input tensor
    vector_length = a_tensor.shape[0]
    
    # Initialize result scalar
    result = nl.ndarray((), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Initialize sum of squares accumulator
    sum_of_squares = nl.zeros((), dtype=nl.float32, buffer=nl.psum)
    
    # Calculate the number of tiles needed
    trip_count = math.ceil(vector_length / nl.tile_size.pmax)
    
    # Process the vector in tiles to respect hardware limitations
    for p in nl.affine_range(trip_count):
        # Calculate indices for the current tile
        start_idx = p * nl.tile_size.pmax
        
        # Create indices for the current tile
        indices = start_idx + nl.arange(nl.tile_size.pmax)
        
        # Load tile with masking to handle boundary conditions
        tile_data = nl.load(a_tensor[indices], mask=(indices < vector_length))
        
        # Square each element in the tile
        squared_tile = nl.square(tile_data)
        
        # Sum the squared values in the tile and accumulate to total sum
        tile_sum = nl.sum(squared_tile, axis=0, mask=(indices < vector_length))
        sum_of_squares += tile_sum
    
    # Calculate the square root of the sum to get the norm
    norm_value = nl.sqrt(sum_of_squares)
    
    # Store the result
    nl.store(result, norm_value)
    
    return result
