from neuronxcc import nki
import neuronxcc.nki.language as nl

@nki.jit
def nki_softmax(input_tensor):
    # Ensure input_tensor is 1D
    if input_tensor.ndim != 1:
        raise ValueError('Input tensor must be 1D.')
        
    # Initialize result array in HBM
    result = nl.ndarray(input_tensor.shape, dtype=input_tensor.dtype, buffer=nl.shared_hbm)

    # Load input to SBUF
    input_sbuf = nl.load(input_tensor)
    
    # Find max value for numerical stability
    max_val = nl.max(input_sbuf, axis=0)
    
    # Shift values by max 
    shifted = nl.subtract(input_sbuf, max_val)
    
    # Compute exp values
    exp_values = nl.exp(shifted)
    
    # Calculate sum of exponentials
    sum_exp = nl.sum(exp_values, axis=0)
    
    # Normalize by dividing by sum
    output_sbuf = nl.divide(exp_values, sum_exp)
    
    # Store result back to HBM
    nl.store(result, output_sbuf)
    
    return result
