import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_prod(a_tensor):
    # Handle scalar input case
    if len(a_tensor.shape) == 0:
        return nl.load(a_tensor)
    
    # Flatten the tensor to make processing simpler
    flattened_shape = (math.prod(a_tensor.shape),)
    
    # Initialize result to 1 with the same dtype as the input tensor
    result_scalar = nl.ones((), dtype=a_tensor.dtype, buffer=nl.psum)
    
    # Calculate the number of tiles needed
    tile_size = nl.tile_size.pmax
    trip_count = math.ceil(flattened_shape[0] / tile_size)
    
    # Process the tensor in tiles
    for p in nl.affine_range(trip_count):
        # Generate tensor indices for the current tile
        i_p = p * tile_size + nl.arange(tile_size)
        
        # Load input data from external memory to on-chip memory
        # Only load up to the actual size of the tensor
        x_tile = nl.load(a_tensor.reshape(flattened_shape)[i_p], mask=(i_p < flattened_shape[0]))
        
        # Compute the product of all elements in this tile
        tile_prod = nl.prod(x_tile, axis=0)
        
        # Accumulate the product into our result
        result_scalar = nl.multiply(result_scalar, tile_prod)
    
    # Create the result tensor with the same dtype as input
    result = nl.ndarray((), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
    
    # Store the result
    nl.store(result, result_scalar)
    
    return result

import neuronxcc.nki.language as nl

    # For scalar input, just return it
    if len(a_tensor.shape) == 0:

    # Flatten the input tensor shape
    flat_size = 1
    for dim in a_tensor.shape:
        flat_size *= dim
    
    # Initialize result to 1 with the same dtype as input
    result = nl.ones((), dtype=a_tensor.dtype)
    
    # Process the tensor in tiles
    tile_size = nl.tile_size.pmax
    num_tiles = math.ceil(flat_size / tile_size)
    
    # Process each tile
    for i in nl.affine_range(num_tiles):
        # Calculate indices for current tile
        start_idx = i * tile_size
        
        # Create range of indices for current tile
        indices = start_idx + nl.arange(tile_size)
        
        # Load tile data with masking to handle boundary cases
        tile_data = nl.load(

        # Compute product for this tile

        # Update running product

    return result
