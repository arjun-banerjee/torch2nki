import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_min(a_tensor, axis=None, keepdims=False):
    """
    Compute the minimum value along the specified axis or axes of a tensor.
    
    Parameters:
    -----------
    a_tensor : ndarray
        The input tensor.
    axis : int or tuple of ints, optional
        The axis or axes along which to find the minimum. Default is None, 
        which means find the minimum over all dimensions.
    keepdims : bool, optional
        If True, the axes which are reduced are left in the result as dimensions with size one.
        
    Returns:
    --------
    result : ndarray
        The minimum value along the specified axis.
    """

    if axis is None:
        # For global minimum, we need to flatten and find min across all elements
        shape = a_tensor.shape
        total_elements = 1
        for dim in shape:
            total_elements *= dim
        
        # Initialize result with first element
        result = nl.zeros((), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
        
        # Calculate how many tiles we need
        trip_count = math.ceil(total_elements / nl.tile_size.pmax)
        first_tile = True
        
        # Process the tensor in tiles to respect hardware limitations
        for p in nl.affine_range(trip_count):
            # Generate tensor indices for the current tile
            i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
            
            # Load input data tile
            valid_indices = i_p < total_elements
            flat_indices = i_p % shape[-1]
            batch_indices = i_p // shape[-1]
            
            if len(shape) == 1:
                x_tile = nl.load(a_tensor[flat_indices], mask=valid_indices)
            elif len(shape) == 2:
                i_p_reshaped = batch_indices[:, None]
                i_f = flat_indices[None, :]
                x_tile = nl.load(a_tensor[i_p_reshaped, i_f], mask=valid_indices)

            tile_min = nl.min(x_tile, axis=0)

            if first_tile:
                nl.store(result, tile_min)
                first_tile = False
            else:
                current_min = nl.load(result)
                new_min = nl.min(nl.stack([current_min, tile_min]))
                nl.store(result, new_min)
        
        return result
