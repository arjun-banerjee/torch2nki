    """

    # Load the input tensors
    v1_tile = nl.load(v1_tensor[i_p])
    v2_tile = nl.load(v2_tensor[i_p])
    
    # Perform element-wise addition using broadcasting
    nl.store(result[i_p], nl.add(v1_tile, v2_tile))

    return result
