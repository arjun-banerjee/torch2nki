def nki_max(a_tensor):
    # Initialize result array

    # Initialize max value in psum buffer
    max_val = nl.zeros((), dtype=a_tensor.dtype, buffer=nl.psum)
    
    # Get tensor shape

    # Iterate over partition axis
    for p in nl.affine_range(sz_p):
        # Load current value
        curr_val = nl.load(a_tensor[p])
        
        # Update max if current value is larger
        max_val = nl.maximum(max_val, curr_val)
    
    # Store final result
    nl.store(result, max_val)
    
    return result
