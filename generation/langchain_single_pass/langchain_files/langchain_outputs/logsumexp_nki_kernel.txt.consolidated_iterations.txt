    result_shape = a_tensor.shape[:-1]
    result = nl.ndarray(result_shape, dtype=a_tensor.dtype, buffer=nl.shared_hbm)

    sz_p, sz_f = a_tensor.shape
    trip_count = math.ceil(sz_p/nl.tile_size.pmax)
    
    # Generate the free dimension index array
    i_f = nl.arange(sz_f)[None, :]
    
    # Process the tensor in tiles
    for p in nl.affine_range(trip_count):
        # Generate tensor indices for the current tile
        i_p = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)[:, None]
        
        # Load input data from external memory to on-chip memory
        x_tile = nl.load(a_tensor[i_p, i_f], mask=(i_p < sz_p))
        
        # Find maximum value along the free dimension
        max_val_tile = nl.maximum.reduce(x_tile, axis=1, keepdims=True)
        
        # Subtract max and compute exp
        exp_tile = nl.exp(x_tile - max_val_tile)
        
        # Sum the exponentials along the free dimension
        sum_exp_tile = nl.sum(exp_tile, axis=1)
        
        # Take log and add max back
        log_sum_exp_tile = nl.log(sum_exp_tile) + max_val_tile[:, 0]
        
        # Store results back to external memory
        nl.store(result[i_p[:, 0]], value=log_sum_exp_tile, mask=(i_p[:, 0] < sz_p))
