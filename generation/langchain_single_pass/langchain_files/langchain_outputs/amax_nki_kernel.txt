***I will implement an amax kernel that accepts both a tensor and an optional dimension parameter, finding the maximum value across the entire tensor (if dim is None) or along a specified dimension, properly handling tiling for large inputs.***

from neuronxcc import nki
import neuronxcc.nki.language as nl
import math

@nki.jit
def nki_amax(a_tensor, dim=None):
    # If dim is specified, we'll reduce along that dimension
    if dim is not None:
        # Handle negative dim indexing
        if dim < 0:
            dim = len(a_tensor.shape) + dim
        
        # Create result array with proper shape
        result_shape = list(a_tensor.shape)
        result_shape.pop(dim)
        if not result_shape:  # If result would be scalar
            result_shape = ()
            
        result = nl.ndarray(tuple(result_shape), dtype=a_tensor.dtype, buffer=nl.shared_hbm)
        
        # Handle scalar result case
        if not result_shape:
            # Flatten tensor for processing
            flat_size = math.prod(a_tensor.shape)
            trip_count = math.ceil(flat_size / nl.tile_size.pmax)
            
            # Initialize max value
            max_val = nl.zeros((), dtype=a_tensor.dtype, buffer=nl.psum)
            
            # Process in tiles
            for p in nl.affine_range(trip_count):
                # Create indices for current tile
                indices = p * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
                
                # Load data with masking for last tile
                tile_data = nl.load(a_tensor.reshape(flat_size)[indices], 
                                   mask=(indices < flat_size))
                
                # Find max in this tile
                tile_max = nl.max(tile_data, axis=0)
                
                # Update global max (first tile or if new max is greater)
                if p == 0:
                    max_val = tile_max
                else:
                    max_val = nl.maximum(max_val, tile_max)
            
            # Store result
            nl.store(result, max_val)
            
        else:
            # For non-scalar results, we need to process each output element
            # Reshape tensor to combine all dimensions except the reduction dimension
            orig_shape = a_tensor.shape
            dim_size = orig_shape[dim]
            
            # Calculate sizes for reshaping
            if dim == 0:
                # Reducing along first dimension
                outer_size = 1
                inner_size = math.prod(orig_shape) // dim_size
                reshaped = a_tensor.reshape((dim_size, inner_size))
                
                # Process in tiles along inner dimension
                inner_tiles = math.ceil(inner_size / nl.tile_size.pmax)
                
                for inner_idx in nl.affine_range(inner_tiles):
                    # Indices for the current inner tile
                    i_inner = inner_idx * nl.tile_size.pmax + nl.arange(nl.tile_size.pmax)
                    i_dim = nl.arange(dim_size)[:, None]
                    
                    # Load data with masking
                    data_tile = nl.load(reshaped[i_dim, i_inner], 
                                      mask=(i_inner < inner_size))
                    
                    # Compute max along reduction dimension
                    max_vals = nl.max(data_tile, axis=0)
                    
                    # Store results
                    nl.store(result.reshape(inner_size)[i_inner], 
                           value=max_vals, 
                           mask=(i_inner < inner_size))
