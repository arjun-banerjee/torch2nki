import neuronxcc.nki.language as nl

@nki.jit
def nki_max(input_tensor):
    # Ensure the input tensor has at least two dimensions
    if len(input_tensor.shape) < 2:
        raise ValueError("Input tensor must have at least two dimensions.")

    # Initialize a result variable to hold the max values
    result = nl.ndarray(input_tensor.shape[:-1], dtype=input_tensor.dtype, buffer=nl.shared_hbm)

    # Compute the maximum value along the last axis (axis=-1)
    max_output = nl.max(nl.transpose(input_tensor), axis=1)

    # Store result back to HBM

    return result
